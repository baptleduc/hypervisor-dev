# One name=value per line, the value runs to the end of the line.
# Only xen_args and dom0_args belong here: env import would let this file
# set any U-Boot variable, load addresses included.

# root= here overrides it.

dom0_args=rw console=hvc0 keep_bootcon bootmem_debug debug dom0_mem=512M
